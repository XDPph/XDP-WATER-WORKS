function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet(); var consumersSheet = sheet.getSheetByName("Consumers"); var billsSheet = sheet.getSheetByName("Readings_Bills"); var name = (e.parameter.name || "").trim().toUpperCase(); var code = (e.parameter.code || "").trim(); var consumersData = consumersSheet.getDataRange().getValues(); var matchedConsumer = null; // Find consumer by Full Name and Account Code for (var i = 1; i < consumersData.length; i++) { var row = consumersData[i]; var acctNo = String(row[0]).trim(); // Account_No var fullName = String(row[1]).trim().toUpperCase(); // Full_Name if (fullName === name && (acctNo === code || (fullName.split(" ")[0] + acctNo) === code)) { matchedConsumer = { accountNo: acctNo, fullName: fullName, purokAddress: row[2], meterNo: row[3], prevReading: row[4] || 0, currReading: row[5] || 0, consumption: (row[5] || 0) - (row[4] || 0), currentCharge: row[6] || 0, balanceDue: row[7] || 0 }; break; } } if (!matchedConsumer) { return ContentService.createTextOutput(JSON.stringify({ success: false })) .setMimeType(ContentService.MimeType.JSON); } // Find matching bills var billsData = billsSheet.getDataRange().getValues(); var matchedBills = []; for (var j = 1; j < billsData.length; j++) { var bRow = billsData[j]; if (String(bRow[2]).trim() === matchedConsumer.accountNo) { matchedBills.push({ billId: bRow[0], billingMonth: bRow[1], paymentDate: bRow[4], prev: bRow[5], curr: bRow[6], cons: bRow[7], charge: bRow[8], arrears: bRow[9], amountDue: bRow[10], status: bRow[11], amountPaid: bRow[12], balanceDue: bRow[13], orNo: bRow[14] }); } } var output = { success: true, consumer: matchedConsumer, bills: matchedBills }; return ContentService.createTextOutput(JSON.stringify(output)) .setMimeType(ContentService.MimeType.JSON); }
